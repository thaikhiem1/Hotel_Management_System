CREATE TABLE IF NOT EXISTS employees (
    employee_id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    full_name VARCHAR(100) NOT NULL,
    position VARCHAR(50) NOT NULL,
    salary NUMERIC(18,2) NOT NULL DEFAULT 0,
    phone VARCHAR(10) NOT NULL UNIQUE,
    citizen_id VARCHAR(12) NOT NULL UNIQUE,
    date_of_birth DATE NOT NULL,
    gender VARCHAR(10) NOT NULL,
    address VARCHAR(255),
    email VARCHAR(100) NOT NULL UNIQUE,
    avatar VARCHAR(255),
    is_active BOOLEAN NOT NULL DEFAULT TRUE,

    CONSTRAINT chk_employee_salary CHECK ( salary >= 0),
    CONSTRAINT chk_gender
        CHECK (gender IN ('Nam', 'Nữ', 'Khác'))
);

CREATE TABLE IF NOT EXISTS room_types (
    room_type_id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    room_type_name VARCHAR(100) NOT NULL UNIQUE,
    number_of_beds INTEGER NOT NULL DEFAULT 1,
    maximum_occupancy INTEGER NOT NULL DEFAULT 2,
    daily_rate NUMERIC(18,2) NOT NULL DEFAULT 0,
    hourly_rate NUMERIC(18,2) NOT NULL DEFAULT 0,

    CONSTRAINT chk_room_type_beds
        CHECK (number_of_beds >= 1),

    CONSTRAINT chk_room_type_occupancy
        CHECK (maximum_occupancy >= 2),

    CONSTRAINT chk_room_type_daily_rate
        CHECK (daily_rate >= 0),

    CONSTRAINT chk_room_type_hourly_rate
        CHECK (hourly_rate >= 0),

      CONSTRAINT chk_room_type_hourly_rate_daily_rate
        CHECK (daily_rate > hourly_rate),
);

CREATE TABLE IF NOT EXISTS rooms (
    room_id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    room_code VARCHAR(20) NOT NULL UNIQUE,
    -- AVAILABLE: Phòng trống
    -- UNDER_MAINTENANCE: Đang sửa chữa
    -- CLEANING: đang dọn dẹp
    room_status VARCHAR(30) NOT NULL DEFAULT 'AVAILABLE',
    room_type_id INTEGER NOT NULL,
    note VARCHAR(255),
    CONSTRAINT fk_room_room_type FOREIGN KEY (room_type_id) REFERENCES room_types(room_type_id),
    CONSTRAINT chk_room_status
        CHECK (room_status IN ('AVAILABLE', 'UNDER_MAINTENANCE', 'CLEANING'))
);

CREATE TABLE IF NOT EXISTS customers (
    customer_id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    full_name VARCHAR(100) NOT NULL,
    phone VARCHAR(10) NOT NULL UNIQUE,
    citizen_id_passport VARCHAR(30) NOT NULL UNIQUE,
    nationality VARCHAR(50) NOT NULL,
    gender VARCHAR(10) NOT NULL,
     CONSTRAINT chk_gender
        CHECK (gender IN ('Nam', 'Nữ', 'Khác'))
);
CREATE TABLE IF NOT EXISTS accounts (
    account_id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    username VARCHAR(50) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    role VARCHAR(30) NOT NULL DEFAULT 'RECEPTIONIST',
    employee_id INTEGER NOT NULL UNIQUE,
    is_active BOOLEAN NOT NULL DEFAULT TRUE,

    CONSTRAINT fk_account_employee
        FOREIGN KEY (employee_id)
        REFERENCES employees(employee_id),

    CONSTRAINT chk_account_role
     CHECK (role IN ('ADMIN','MANAGER' , 'RECEPTIONIST'))
);
CREATE TABLE IF NOT EXISTS service_categories (
    service_category_id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    category_name VARCHAR(100) NOT NULL UNIQUE
);
CREATE TABLE IF NOT EXISTS services (
    service_id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    service_name VARCHAR(100) NOT NULL UNIQUE,
    unit_price NUMERIC(18,2) NOT NULL DEFAULT 0,
    stock_quantity INTEGER,
    unit VARCHAR(20) NOT NULL,
    service_category_id INTEGER NOT NULL,

    CONSTRAINT fk_service_category
        FOREIGN KEY (service_category_id)
        REFERENCES service_categories(service_category_id),

    CONSTRAINT chk_service_unit_price
        CHECK (unit_price >= 0),

    CONSTRAINT chk_service_stock
        CHECK (stock_quantity IS NULL OR stock_quantity >= 0)
);

CREATE TABLE IF NOT EXISTS rental_receipts (
    rental_receipt_id      INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    -- Mã phiếu duy nhất, do hệ thống sinh (VC-13)
    receipt_code           VARCHAR(20) NOT NULL UNIQUE,
    rental_date            TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    customer_id            INTEGER NOT NULL,   -- khách hàng đại diện
    employee_id            INTEGER NOT NULL,   -- nhân viên lập phiếu

    -- DA_XAC_NHAN: đã xác nhận | DANG_LUU_TRU: đang lưu trú
    -- DA_TRA_PHONG: đã trả phòng | DA_HUY: đã hủy
    receipt_status         VARCHAR(20) NOT NULL DEFAULT 'DA_XAC_NHAN',

    -- Tổng chi phí dự kiến = SUM(booking_details.estimated_amount) (BR-REN-001-10)
    estimated_total_amount NUMERIC(18,2) NOT NULL DEFAULT 0,

    -- Lịch sử hủy toàn bộ phiếu (UC-REN-003)
    cancelled_at           TIMESTAMP,
    cancelled_by           INTEGER,
    cancel_reason          VARCHAR(255),

    version                INTEGER NOT NULL DEFAULT 0,

    CONSTRAINT fk_rental_receipt_customer FOREIGN KEY (customer_id)
        REFERENCES customers(customer_id),
    CONSTRAINT fk_rental_receipt_employee FOREIGN KEY (employee_id)
        REFERENCES employees(employee_id),
    CONSTRAINT fk_rental_receipt_cancelled_by FOREIGN KEY (cancelled_by)
        REFERENCES employees(employee_id),

    CONSTRAINT chk_receipt_status
        CHECK (receipt_status IN ('DA_XAC_NHAN', 'DANG_LUU_TRU', 'DA_TRA_PHONG', 'DA_HUY')),
    CONSTRAINT chk_receipt_estimated_total CHECK (estimated_total_amount >= 0),
    CONSTRAINT chk_receipt_cancel_info CHECK (
        (receipt_status = 'DA_HUY'  AND cancelled_at IS NOT NULL)
     OR (receipt_status <> 'DA_HUY' AND cancelled_at IS NULL)
    )
);

CREATE TABLE IF NOT EXISTS booking_details (
    booking_detail_id       INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    rental_receipt_id       INTEGER NOT NULL,
    room_id                 INTEGER NOT NULL,

    number_of_guests        INTEGER NOT NULL,   -- dự kiến
    actual_number_of_guests INTEGER,            -- ghi nhận khi check-in (UC-REN-004)

    -- Thời gian dự kiến
    check_in_at             TIMESTAMP NOT NULL,
    check_out_at            TIMESTAMP NOT NULL,
    -- Thời gian thực tế
    actual_check_in_at      TIMESTAMP,
    actual_check_out_at     TIMESTAMP,

    -- RESERVED: đã đặt | CHECKED_IN: đang thuê
    -- CHECKED_OUT: đã trả phòng | CANCELLED: đã hủy
    booking_status          VARCHAR(20) NOT NULL DEFAULT 'RESERVED',

    -- Phương thức tính giá: NGAY (theo đêm) | GIO (theo giờ)
    pricing_method          VARCHAR(10) NOT NULL,
    -- Đơn giá đã chốt tại thời điểm tạo phiếu (BR-REN-001-09)
    unit_price              NUMERIC(18,2) NOT NULL,
    -- Số đêm/giờ tính tiền dự kiến và thành tiền dự kiến (UC-REN-010)
    estimated_units         INTEGER NOT NULL,
    estimated_amount        NUMERIC(18,2) NOT NULL,

    -- Lịch sử hủy chi tiết (UC-REN-002)
    cancelled_at            TIMESTAMP,
    cancelled_by            INTEGER,
    cancel_reason           VARCHAR(255),

    version                 INTEGER NOT NULL DEFAULT 0,

    CONSTRAINT fk_booking_detail_rental_receipt FOREIGN KEY (rental_receipt_id)
        REFERENCES rental_receipts(rental_receipt_id),
    CONSTRAINT fk_booking_detail_room FOREIGN KEY (room_id)
        REFERENCES rooms(room_id),
    CONSTRAINT fk_booking_detail_cancelled_by FOREIGN KEY (cancelled_by)
        REFERENCES employees(employee_id),

    CONSTRAINT chk_booking_number_of_guests CHECK (number_of_guests > 0),
    CONSTRAINT chk_booking_actual_guests
        CHECK (actual_number_of_guests IS NULL OR actual_number_of_guests > 0),
    CONSTRAINT chk_booking_date CHECK (check_out_at > check_in_at),

    -- Không nhận phòng trước giờ dự kiến (UC-REN-004)
    CONSTRAINT chk_actual_check_in
        CHECK (actual_check_in_at IS NULL OR actual_check_in_at >= check_in_at),
    CONSTRAINT chk_actual_check_out
        CHECK (actual_check_out_at IS NULL
               OR (actual_check_in_at IS NOT NULL AND actual_check_out_at >= actual_check_in_at)),

    CONSTRAINT chk_booking_pricing_method CHECK (pricing_method IN ('NGAY', 'GIO')),
    CONSTRAINT chk_booking_unit_price CHECK (unit_price >= 0),
    CONSTRAINT chk_booking_estimated_units CHECK (estimated_units > 0),
    CONSTRAINT chk_booking_estimated_amount
        CHECK (estimated_amount >= 0 AND estimated_amount = ROUND(unit_price * estimated_units, 2)),

    CONSTRAINT chk_booking_status
        CHECK (booking_status IN ('RESERVED', 'CHECKED_IN', 'CHECKED_OUT', 'CANCELLED')),

    -- Dữ liệu thời gian/hủy phải khớp với trạng thái
    CONSTRAINT chk_booking_status_consistency CHECK (
        (booking_status = 'RESERVED'
            AND actual_check_in_at IS NULL AND actual_check_out_at IS NULL AND cancelled_at IS NULL)
     OR (booking_status = 'CHECKED_IN'
            AND actual_check_in_at IS NOT NULL AND actual_number_of_guests IS NOT NULL
            AND actual_check_out_at IS NULL AND cancelled_at IS NULL)
     OR (booking_status = 'CHECKED_OUT'
            AND actual_check_in_at IS NOT NULL AND actual_check_out_at IS NOT NULL
            AND cancelled_at IS NULL)
     OR (booking_status = 'CANCELLED'
            AND actual_check_in_at IS NULL AND cancelled_at IS NOT NULL)
    ),

    -- Không chồng lấn lịch của cùng một phòng với các lượt còn hiệu lực.
    -- Range '[)' nên hai khoảng tiếp giáp không bị xem là chồng lấn (BR-REN-009-04).
    -- CANCELLED và CHECKED_OUT không giữ chỗ (BR-REN-009-09).
    CONSTRAINT excl_booking_room_overlap EXCLUDE USING gist (
        room_id WITH =,
        tsrange(check_in_at, check_out_at, '[)') WITH &&
    ) WHERE (booking_status IN ('RESERVED', 'CHECKED_IN'))
);

-- Mỗi phòng chỉ xuất hiện một lần trong cùng phiếu thuê (BR-REN-001-04)
CREATE UNIQUE INDEX IF NOT EXISTS uq_booking_receipt_room_active
    ON booking_details (rental_receipt_id, room_id)
    WHERE booking_status <> 'CANCELLED';

-- Một phòng chỉ có tối đa một lượt đang lưu trú (UC-REN-004 E4, E5)
CREATE UNIQUE INDEX IF NOT EXISTS uq_booking_room_checked_in
    ON booking_details (room_id)
    WHERE booking_status = 'CHECKED_IN';

-- 10. BOOKING DETAIL SERVICES (dịch vụ khách sử dụng - UC-SVC-005)
-- Dùng khóa thay thế: cùng một dịch vụ có thể được ghi nhận nhiều lần
-- trong một lượt lưu trú, mỗi lần với đơn giá đã chốt riêng (BR9).
-- =========================================================
CREATE TABLE IF NOT EXISTS booking_detail_services (
    booking_detail_service_id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    booking_detail_id         INTEGER NOT NULL,
    service_id                INTEGER NOT NULL,
    employee_id               INTEGER NOT NULL,   -- RECEPTIONIST ghi nhận
    used_at                   TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,

    unit_price                NUMERIC(18,2) NOT NULL,   -- đơn giá đã chốt
    quantity                  INTEGER NOT NULL,
    total_amount              NUMERIC(18,2) NOT NULL,

    CONSTRAINT fk_bds_booking_detail FOREIGN KEY (booking_detail_id)
        REFERENCES booking_details(booking_detail_id),
    CONSTRAINT fk_bds_service FOREIGN KEY (service_id)
        REFERENCES services(service_id),
    CONSTRAINT fk_bds_employee FOREIGN KEY (employee_id)
        REFERENCES employees(employee_id),

    CONSTRAINT chk_bds_unit_price CHECK (unit_price >= 0),
    CONSTRAINT chk_bds_quantity   CHECK (quantity > 0),
    -- Thành tiền = số lượng x đơn giá áp dụng (BR7)
    CONSTRAINT chk_bds_total CHECK (total_amount = ROUND(unit_price * quantity, 2))
);

-- 11. INVOICES
-- Mỗi chi tiết lưu trú tối đa một hóa đơn (UC-REN-007 BR3)
-- =========================================================
CREATE TABLE IF NOT EXISTS invoices (
    invoice_id          INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    invoice_code        VARCHAR(20) NOT NULL UNIQUE,
    invoice_date        TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,

    booking_detail_id   INTEGER NOT NULL UNIQUE,
    employee_id         INTEGER NOT NULL,   -- nhân viên trả phòng / lập hóa đơn

    -- Số liệu đã chốt khi trả phòng, không tính lại khi xem/xuất
    billed_units        INTEGER NOT NULL,          -- số đêm/giờ thực tế tính tiền
    room_amount         NUMERIC(18,2) NOT NULL,
    service_amount      NUMERIC(18,2) NOT NULL DEFAULT 0,
    total_amount        NUMERIC(18,2) NOT NULL,

    payment_status      VARCHAR(10) NOT NULL DEFAULT 'UNPAID',
    paid_at             TIMESTAMP,                 -- NULL khi chưa thanh toán
    paid_by             INTEGER,                   -- nhân viên xác nhận thanh toán

    version             INTEGER NOT NULL DEFAULT 0,

    CONSTRAINT fk_invoice_booking_detail FOREIGN KEY (booking_detail_id)
        REFERENCES booking_details(booking_detail_id),
    CONSTRAINT fk_invoice_employee FOREIGN KEY (employee_id)
        REFERENCES employees(employee_id),
    CONSTRAINT fk_invoice_paid_by FOREIGN KEY (paid_by)
        REFERENCES employees(employee_id),

    CONSTRAINT chk_invoice_billed_units   CHECK (billed_units > 0),
    CONSTRAINT chk_invoice_room_amount    CHECK (room_amount >= 0),
    CONSTRAINT chk_invoice_service_amount CHECK (service_amount >= 0),
    -- T_cuối = T_phòng + T_dịch vụ (UC-REN-007)
    CONSTRAINT chk_invoice_total CHECK (total_amount = room_amount + service_amount),

    CONSTRAINT chk_invoice_payment_status CHECK (payment_status IN ('UNPAID', 'PAID')),
    CONSTRAINT chk_invoice_payment_time CHECK (
        (payment_status = 'UNPAID' AND paid_at IS NULL     AND paid_by IS NULL)
     OR (payment_status = 'PAID'   AND paid_at IS NOT NULL AND paid_by IS NOT NULL)
    ),
    CONSTRAINT chk_invoice_paid_after_created
        CHECK (paid_at IS NULL OR paid_at >= invoice_date)
);

CREATE TABLE monthly_revenue_statistics (

    last_updated_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT uq_monthly_revenue
        UNIQUE (year, month),

    CONSTRAINT chk_revenue_month
        CHECK (month BETWEEN 1 AND 12),

    CONSTRAINT chk_revenue_year
        CHECK (year > 0),

    CONSTRAINT chk_total_revenue
        CHECK (total_revenue >= 0),

    CONSTRAINT chk_invoice_count
        CHECK (invoice_count >= 0)
);
