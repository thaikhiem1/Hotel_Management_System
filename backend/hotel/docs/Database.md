-- =========================================================
-- 1. EMPLOYEE
-- =========================================================

CREATE TABLE employees (
    employee_id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    full_name VARCHAR(100) NOT NULL,
    position VARCHAR(50),
    salary NUMERIC(18,2),
    phone VARCHAR(15),
    citizen_id VARCHAR(20) UNIQUE,
    date_of_birth DATE,
    gender VARCHAR(10),
    address VARCHAR(255),
    email VARCHAR(100),
    avatar VARCHAR(255),
    is_active BOOLEAN NOT NULL DEFAULT TRUE,

    CONSTRAINT chk_employee_salary
        CHECK (salary IS NULL OR salary >= 0)
);


-- =========================================================
-- 2. ACCOUNT
-- =========================================================

CREATE TABLE accounts (
    username VARCHAR(50) PRIMARY KEY,
    password_hash VARCHAR(255) NOT NULL,

    -- 0: Admin
    -- 1: Receptionist
    -- 2: Manager
    role INTEGER NOT NULL,

    employee_id INTEGER NOT NULL UNIQUE,
    is_active BOOLEAN NOT NULL DEFAULT TRUE,

    CONSTRAINT fk_account_employee
        FOREIGN KEY (employee_id)
        REFERENCES employees(employee_id),

    CONSTRAINT chk_account_role
        CHECK (role IN (0, 1, 2))
);


-- =========================================================
-- 3. CUSTOMER
-- =========================================================

CREATE TABLE customers (
    customer_id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    full_name VARCHAR(100) NOT NULL,
    phone VARCHAR(15),
    citizen_id_passport VARCHAR(30),
    nationality VARCHAR(50),
    gender VARCHAR(10)
);


-- =========================================================
-- 4. ROOM TYPE
-- =========================================================

CREATE TABLE room_types (
    room_type_id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    room_type_name VARCHAR(100) NOT NULL,
    number_of_beds INTEGER NOT NULL,
    maximum_occupancy INTEGER NOT NULL,
    daily_rate NUMERIC(18,2) NOT NULL,
    hourly_rate NUMERIC(18,2),

    CONSTRAINT chk_room_type_beds
        CHECK (number_of_beds > 0),

    CONSTRAINT chk_room_type_occupancy
        CHECK (maximum_occupancy > 0),

    CONSTRAINT chk_room_type_daily_rate
        CHECK (daily_rate >= 0),

    CONSTRAINT chk_room_type_hourly_rate
        CHECK (hourly_rate IS NULL OR hourly_rate >= 0)
);


-- =========================================================
-- 5. ROOM
-- =========================================================

CREATE TABLE rooms (
    room_id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    room_code VARCHAR(20) NOT NULL UNIQUE,

    -- AVAILABLE: Phòng trống
    -- UNDER_MAINTENANCE: Đang sửa chữa
    room_status VARCHAR(30) NOT NULL DEFAULT 'AVAILABLE',

    room_type_id INTEGER NOT NULL,
    note VARCHAR(255),

    CONSTRAINT fk_room_room_type
        FOREIGN KEY (room_type_id)
        REFERENCES room_types(room_type_id),

    CONSTRAINT chk_room_status
        CHECK (room_status IN ('AVAILABLE', 'UNDER_MAINTENANCE'))
);


-- =========================================================
-- 6. RENTAL RECEIPT
-- =========================================================

CREATE TABLE rental_receipts (
    rental_receipt_id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    rental_date DATE NOT NULL,
    customer_id INTEGER NOT NULL,
    employee_id INTEGER NOT NULL,

    CONSTRAINT fk_rental_receipt_customer
        FOREIGN KEY (customer_id)
        REFERENCES customers(customer_id),

    CONSTRAINT fk_rental_receipt_employee
        FOREIGN KEY (employee_id)
        REFERENCES employees(employee_id)
);


-- =========================================================
-- 7. BOOKING DETAIL
-- =========================================================

CREATE TABLE booking_details (
    booking_detail_id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    rental_receipt_id INTEGER NOT NULL,
    room_id INTEGER NOT NULL,

    number_of_guests INTEGER NOT NULL,
    check_in_date DATE NOT NULL,
    check_out_date DATE NOT NULL,

    -- RESERVED: Đã đặt
    -- CHECKED_IN: Đang thuê
    -- CLEANING: Đang dọn dẹp
    booking_status VARCHAR(30) NOT NULL,

    unit_price NUMERIC(18,2) NOT NULL,
    total_amount NUMERIC(18,2) NOT NULL,

    CONSTRAINT fk_booking_detail_rental_receipt
        FOREIGN KEY (rental_receipt_id)
        REFERENCES rental_receipts(rental_receipt_id),

    CONSTRAINT fk_booking_detail_room
        FOREIGN KEY (room_id)
        REFERENCES rooms(room_id),

    CONSTRAINT chk_booking_number_of_guests
        CHECK (number_of_guests > 0),

    CONSTRAINT chk_booking_date
        CHECK (check_out_date > check_in_date),

    CONSTRAINT chk_booking_unit_price
        CHECK (unit_price >= 0),

    CONSTRAINT chk_booking_total_amount
        CHECK (total_amount >= 0),

    CONSTRAINT chk_booking_status
        CHECK (
            booking_status IN (
                'RESERVED',
                'CHECKED_IN',
                'CLEANING'
            )
        )
);


-- =========================================================
-- 8. SERVICE CATEGORY
-- =========================================================

CREATE TABLE service_categories (
    service_category_id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    category_name VARCHAR(100) NOT NULL
);


-- =========================================================
-- 9. SERVICE
-- =========================================================

CREATE TABLE services (
    service_id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    service_name VARCHAR(100) NOT NULL,
    unit_price NUMERIC(18,2) NOT NULL,
    stock_quantity INTEGER,
    unit VARCHAR(20) NOT NULL,
    service_category_id INTEGER NOT NULL,

    CONSTRAINT fk_service_category
        FOREIGN KEY (service_category_id)
        REFERENCES service_categories(service_category_id),

    CONSTRAINT chk_service_unit_price
        CHECK (unit_price >= 0),

    CONSTRAINT chk_service_stock
        CHECK (stock_quantity IS NULL OR stock_quantity >= 0)
);


-- =========================================================
-- 10. BOOKING DETAIL - SERVICE
-- =========================================================

CREATE TABLE booking_detail_services (
    booking_detail_id INTEGER NOT NULL,
    service_id INTEGER NOT NULL,

    unit_price NUMERIC(18,2) NOT NULL,
    quantity INTEGER NOT NULL,
    total_amount NUMERIC(18,2) NOT NULL,

    PRIMARY KEY (booking_detail_id, service_id),

    CONSTRAINT fk_booking_detail_service_booking
        FOREIGN KEY (booking_detail_id)
        REFERENCES booking_details(booking_detail_id),

    CONSTRAINT fk_booking_detail_service_service
        FOREIGN KEY (service_id)
        REFERENCES services(service_id),

    CONSTRAINT chk_booking_service_unit_price
        CHECK (unit_price >= 0),

    CONSTRAINT chk_booking_service_quantity
        CHECK (quantity > 0),

    CONSTRAINT chk_booking_service_total
        CHECK (total_amount >= 0)
);


-- =========================================================
-- 11. INVOICE
-- =========================================================

CREATE TABLE invoices (
    invoice_id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    invoice_date DATE NOT NULL,

    total_amount NUMERIC(18,2) NOT NULL,

    employee_id INTEGER NOT NULL,
    rental_receipt_id INTEGER NOT NULL UNIQUE,

    -- UNPAID: Chưa thanh toán
    -- PAID: Đã thanh toán
    payment_status VARCHAR(20) NOT NULL,

    CONSTRAINT fk_invoice_employee
        FOREIGN KEY (employee_id)
        REFERENCES employees(employee_id),

    CONSTRAINT fk_invoice_rental_receipt
        FOREIGN KEY (rental_receipt_id)
        REFERENCES rental_receipts(rental_receipt_id),

    CONSTRAINT chk_invoice_total_amount
        CHECK (total_amount >= 0),

    CONSTRAINT chk_invoice_payment_status
        CHECK (
            payment_status IN ('UNPAID', 'PAID')
        )
);


-- =========================================================
-- 12. MONTHLY REVENUE STATISTICS
-- =========================================================

CREATE TABLE monthly_revenue_statistics (
    statistic_id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    year INTEGER NOT NULL,
    month INTEGER NOT NULL,

    total_revenue NUMERIC(18,2) NOT NULL,
    invoice_count INTEGER NOT NULL,

    last_updated_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT uq_monthly_revenue
        UNIQUE (year, month),

    CONSTRAINT chk_revenue_month
        CHECK (month BETWEEN 1 AND 12),

    CONSTRAINT chk_revenue_year
        CHECK (year > 0),

    CONSTRAINT chk_total_revenue
        CHECK (total_revenue >= 0),

    CONSTRAINT chk_invoice_count
        CHECK (invoice_count >= 0)
);
-- Account -> Employee
-- 1 Employee <-> 0..1 Account
-- employee_id UNIQUE trong accounts

-- Room -> Room Type
-- N Rooms -> 1 Room Type

-- Rental Receipt -> Customer
-- N Rental Receipts -> 1 Customer

-- Rental Receipt -> Employee
-- N Rental Receipts -> 1 Employee

-- Booking Detail -> Rental Receipt
-- N Booking Details -> 1 Rental Receipt

-- Booking Detail -> Room
-- N Booking Details -> 1 Room

-- Service -> Service Category
-- N Services -> 1 Service Category

-- Booking Detail <-> Service
-- Many-to-Many through booking_detail_services

-- Invoice -> Rental Receipt
-- 1 Rental Receipt <-> 0..1 Invoice

-- Invoice -> Employee
-- N Invoices -> 1 Employee
